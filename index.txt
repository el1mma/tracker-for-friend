const KEY = 'goalquest_friend_v1';

// ПУСТОЙ СТАРТ — никаких шаблонных целей
const SEED = [];

function makeSeedGoals() {
  return SEED.map((s, idx) => {
    const g = Object.assign({}, s);
    g.id = Date.now() + idx;
    g.done = false;
    g.lastLog = null;
    if (s.type === 'weight') { g.current = s.current; g.target = s.target; }
    else g.current = 0;
    return g;
  });
}

const goalsEl = document.getElementById('goals');
let state;
try { state = JSON.parse(localStorage.getItem(KEY)); } catch(e) { state = null; }
if (!state || !Array.isArray(state.goals)) {
  state = { goals: makeSeedGoals(), lastBackup: new Date().toISOString() };
  localStorage.setItem(KEY, JSON.stringify(state));
}

function save() { localStorage.setItem(KEY, JSON.stringify(state)); render(); }
function isImage(s) { return /^https?:\/\//.test(s); }
function today() { return new Date().toISOString().slice(0, 10); }

function toggleType() {
  const t = document.getElementById('gType').value;
  document.getElementById('countFields').style.display  = t === 'count' ? 'block' : 'none';
  document.getElementById('weightFields').style.display = t === 'weight' ? 'block' : 'none';
}

/* ---------- СВОРАЧИВАНИЕ ФОРМЫ ---------- */
function toggleNewGoal(forceState) {
  const form = document.getElementById('newGoalForm');
  const arrow = document.getElementById('newGoalArrow');
  const isHidden = form.style.display === 'none' || form.style.display === '';
  const newState = forceState !== undefined ? forceState : isHidden;
  
  form.style.display = newState ? 'block' : 'none';
  if (newState) arrow.classList.add('rotated');
  else arrow.classList.remove('rotated');
}

/* ---------- КОНФЕТТИ ---------- */
function checkAndCelebrate(oldFrac, newFrac, isFinal) {
  const tiers = [1/3, 2/3, 1.0];
  let unlockedTier = 0;
  for (let i = 0; i < tiers.length; i++) {
    if (oldFrac < tiers[i] && newFrac >= tiers[i]) unlockedTier = i + 1;
  }
  if (unlockedTier === 3 || isFinal) {
    var end = Date.now() + 2500;
    (function frame() {
      confetti({ particleCount: 6, angle: 60, spread: 60, origin: { x: 0, y: 0.7 }, colors: ['#f59e0b','#10b981','#3b82f6','#fbbf24'] });
      confetti({ particleCount: 6, angle: 120, spread: 60, origin: { x: 1, y: 0.7 }, colors: ['#f59e0b','#10b981','#3b82f6','#fbbf24'] });
      if (Date.now() < end) requestAnimationFrame(frame);
    }());
  } else if (unlockedTier > 0) {
    confetti({ particleCount: 120, spread: 80, origin: { y: 0.7 }, colors: ['#f59e0b','#fbbf24','#10b981','#3b82f6'], gravity: 0.8, scalar: 1.2 });
  }
}

const toast = document.getElementById('toast');
const toastMsg = document.getElementById('toastMsg');
const toastIco = document.getElementById('toastIco');
const toastActions = document.getElementById('toastActions');
let toastTimer = null;

function showToast(ico, msg, actions) {
  toastIco.textContent = ico;
  toastMsg.textContent = msg;
  toastActions.innerHTML = '';
  (actions || []).forEach(a => {
    const b = document.createElement('button');
    b.className = 'btn-ghost' + (a.primary ? ' primary' : '');
    b.textContent = a.label;
    b.onclick = a.onClick;
    toastActions.appendChild(b);
  });
  toast.classList.add('show');
  clearTimeout(toastTimer);
  if (!actions) toastTimer = setTimeout(hideToast, 4000);
}
function hideToast() { toast.classList.remove('show'); }

function daysSince(iso) {
  if (!iso) return Infinity;
  return (Date.now() - new Date(iso).getTime()) / 86400000;
}

function checkBackupReminder() {
  if (daysSince(state.lastBackup) > 7) {
    showToast('💾', 'Прошло больше недели с последнего бэкапа. Сохрани прогресс, чтобы не потерять его.', [
      { label: 'Скачать сейчас', primary: true, onClick: () => { exportBackup(); } },
      { label: 'Позже', onClick: () => { hideToast(); } }
    ]);
  }
}

/* ---------- БЭКАП ---------- */
function markBackup() {
  state.lastBackup = new Date().toISOString();
  localStorage.setItem(KEY, JSON.stringify(state));
}

function exportBackup() {
  const d = new Date().toISOString().slice(0, 10);
  const blob = new Blob([JSON.stringify(state, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'goal-tracker-backup-' + d + '.json';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
  markBackup();
  hideToast();
  showToast('✅', 'Бэкап сохранён. Перешли файл в Telegram «Избранное» или облако.');
}

document.getElementById('importFile').addEventListener('change', function(e) {
  const file = e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(ev) {
    let parsed;
    try { parsed = JSON.parse(ev.target.result); }
    catch(err) { alert('Файл повреждён или это не бэкап трекера.'); e.target.value=''; return; }
    if (!parsed || !Array.isArray(parsed.goals)) { alert('В файле нет данных целей.'); e.target.value=''; return; }
    if (!confirm('Загрузить этот бэкап? Текущий прогресс будет заменён.')) { e.target.value=''; return; }
    state = parsed;
    if (!state.lastBackup) state.lastBackup = new Date().toISOString();
    save();
    markBackup();
    e.target.value = '';
    showToast('📂', 'Прогресс восстановлен из бэкапа.');
  };
  reader.readAsText(file);
});

/* ---------- ДОБАВЛЕНИЕ ---------- */
function addGoal() {
  const type = document.getElementById('gType').value;
  const name = document.getElementById('gName').value.trim();
  if (!name) { alert('Введите название цели'); return; }

  const rewards = [1,2,3].map(n => {
    const icon = document.getElementById('r' + n + 'icon').value.trim();
    const rname = document.getElementById('r' + n + 'name').value.trim();
    const rimg = document.getElementById('r' + n + 'img').value.trim();
    return { icon: icon || '🎁', name: rname || ('Этап ' + n), img: rimg || '' };
  });

  if (type === 'count') {
    const totalRaw = document.getElementById('gTotal').value.trim();
    const total = parseFloat(totalRaw);
    const unit = document.getElementById('gUnit').value.trim();
    if (totalRaw === '' || isNaN(total) || total <= 0) { alert('Введите общее количество числом'); return; }
    if (!unit) { alert('Введите единицу (дней / шт)'); return; }
    state.goals.push({ id: Date.now(), type: 'count', name, total, unit, current: 0, rewards, done: false, lastLog: null });
  } else {
    const start = parseFloat(document.getElementById('gStart').value);
    const lose = parseFloat(document.getElementById('gLose').value);
    if (isNaN(start) || isNaN(lose) || start <= 0 || lose <= 0) { alert('Введите стартовый вес и сколько сбросить'); return; }
    if (lose >= start) { alert('Сбросить больше стартового веса нельзя'); return; }
    state.goals.push({ id: Date.now(), type: 'weight', name, start, lose, target: start - lose, current: start, rewards, done: false, lastLog: null });
  }

  ['gName','gTotal','gUnit','gStart','gLose',
   'r1icon','r1name','r1img','r2icon','r2name','r2img','r3icon','r3name','r3img']
    .forEach(id => document.getElementById(id).value = '');
  
  // Возвращаем иконки по умолчанию 🎁
  ['r1icon','r2icon','r3icon'].forEach(id => document.getElementById(id).value = '🎁');

  save();
  setTimeout(() => goalsEl.scrollIntoView({ behavior: 'smooth', block: 'start' }), 100);
  
  toggleNewGoal(false);
}

/* ---------- ПРОГРЕСС ---------- */
function getLost(g) { return g.start - g.current; }
function progressFraction(g) {
  if (g.type === 'weight') return Math.min(1, getLost(g) / g.lose);
  return Math.min(1, g.current / g.total);
}

function addCount(id) {
  const g = state.goals.find(x => x.id === id);
  if (g.done) return;
  if (g.lastLog === today()) { alert('Сегодня уже отмечено. Приходите завтра!'); return; }
  const oldFrac = progressFraction(g);
  g.current += 1;
  if (g.current >= g.total) { g.current = g.total; g.done = true; }
  g.lastLog = today();
  checkAndCelebrate(oldFrac, progressFraction(g), g.done);
  save();
}

function saveWeight(id) {
  const g = state.goals.find(x => x.id === id);
  if (g.done) return;
  const inp = document.getElementById('w_' + id);
  const v = parseFloat(inp.value);
  if (isNaN(v) || v <= 0) { alert('Введите текущий вес числом'); return; }
  if (g.lastLog === today()) { alert('Сегодня уже отмечено. Приходите завтра!'); return; }
  if (v >= g.current) { alert('Вес должен быть меньше предыдущего (' + g.current + ' кг)'); return; }
  const oldFrac = progressFraction(g);
  g.current = v;
  if (g.current <= g.target) { g.current = g.target; g.done = true; }
  g.lastLog = today();
  checkAndCelebrate(oldFrac, progressFraction(g), g.done);
  save();
}

function resetGoal(id) {
  const g = state.goals.find(x => x.id === id);
  if (!confirm('Сбросить прогресс цели «' + g.name + '»?')) return;
  g.current = g.type === 'weight' ? g.start : 0;
  g.done = false; g.lastLog = null;
  closeAllMenus(); save();
}
function delGoal(id) {
  const g = state.goals.find(x => x.id === id);
  if (!confirm('Удалить цель «' + g.name + '» навсегда?')) return;
  state.goals = state.goals.filter(x => x.id !== id);
  closeAllMenus(); save();
}
function resetAll() {
  if (!confirm('Сбросить ВСЁ? Все цели и прогресс будут удалены.')) return;
  state = { goals: [], lastBackup: new Date().toISOString() }; save();
}

/* ---------- МЕНЮ ---------- */
function toggleMenu(id) {
  const menu = document.getElementById('menu-' + id);
  const shown = menu.classList.contains('show');
  closeAllMenus();
  if (!shown) menu.classList.add('show');
}
function closeAllMenus() {
  document.querySelectorAll('.dropdown-menu').forEach(m => m.classList.remove('show'));
}
document.addEventListener('click', e => {
  if (!e.target.closest('.menu-btn') && !e.target.closest('.dropdown-menu')) closeAllMenus();
});

/* ---------- ОТРИСОВКА ---------- */
function render() {
  if (!state.goals.length) {
    goalsEl.innerHTML = `
      <div class="empty-state">
        <span class="emoji">🎁</span>
        <div class="title">Пока нет целей</div>
        <div>Нажми «➕ Новая цель» выше, чтобы добавить первую!</div>
      </div>`;
    return;
  }

  goalsEl.innerHTML = state.goals.map(g => {
    const frac = progressFraction(g);
    const pct = (frac * 100).toFixed(0);
    const third = 1/3;

    let progText;
    if (g.type === 'weight') {
      progText = `<strong>${g.current}</strong> кг <span style="color:var(--text-secondary)">(сброшено ${getLost(g).toFixed(1)} из ${g.lose} кг)</span> · ${pct}%`;
    } else {
      progText = `<strong>${g.current}</strong> / ${g.total} ${g.unit} · ${pct}%`;
    }

    const slots = g.rewards.map((r, i) => {
      const needFrac = third * (i + 1);
      const unlocked = frac >= needFrac - 1e-9;

      let stageLabel;
      if (g.type === 'weight') stageLabel = '−' + (g.lose * needFrac).toFixed(1) + ' кг';
      else stageLabel = Math.round(g.total * needFrac) + ' ' + g.unit;

      let visual;
      if (isImage(r.img)) {
        visual = `<img src="${r.img}" class="reward-visual" alt="${r.name}"
                    onerror="this.style.display='none'; this.nextElementSibling.style.display='block';">
                  <div class="reward-emoji" style="display:none">${r.icon}</div>`;
      } else {
        visual = `<div class="reward-emoji">${r.icon}</div>`;
      }

      return `<div class="reward-slot ${unlocked ? 'unlocked' : 'locked'}">
        <div class="reward-stage">${stageLabel}</div>
        ${visual}
        <div class="reward-name">${r.name}</div>
      </div>`;
    }).join('');

    let controls;
    if (g.done) {
      controls = '<div class="frozen">🏆 Цель выполнена! Прогресс заморожен.</div>';
    } else if (g.type === 'count') {
      const blocked = g.lastLog === today();
      controls = `<button class="btn-primary" onclick="addCount(${g.id})" ${blocked ? 'disabled' : ''}>
        ${blocked ? '✅ Отмечено сегодня' : '+1 ' + g.unit}
      </button>`;
    } else {
      const blocked = g.lastLog === today();
      controls = `<div class="weight-box">
        <input id="w_${g.id}" type="number" step="0.1" placeholder="Вес сегодня" ${blocked ? 'disabled' : ''}>
        <button class="btn-primary btn-small" onclick="saveWeight(${g.id})" ${blocked ? 'disabled' : ''}>
          ${blocked ? '✅' : 'OK'}
        </button>
      </div>
      ${blocked ? '<div class="hint" style="text-align:center;margin-top:8px;">Сегодня уже отмечено</div>' : ''}`;
    }

    return `<div class="card goal ${g.done ? 'done' : ''}">
      <div class="goal-header">
        <div class="goal-title">${g.name}${g.done ? ' <span class="cup">🏆</span>' : ''}</div>
        <button class="menu-btn" onclick="toggleMenu(${g.id})">⋮</button>
        <div class="dropdown-menu" id="menu-${g.id}">
          <button onclick="resetGoal(${g.id})">↺ Сбросить прогресс</button>
          <button class="danger" onclick="delGoal(${g.id})">🗑 Удалить цель</button>
        </div>
      </div>
      <div class="xp-bar"><div class="xp-fill" style="width:${pct}%"></div></div>
      <div class="progress-text">${progText}</div>
      <div class="rewards">${slots}</div>
      <div style="margin-top:20px">${controls}</div>
    </div>`;
  }).join('');
}
